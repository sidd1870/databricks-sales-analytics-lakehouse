
### Retrieve Valid Product Hierarchy
```sql
SELECT category, subcategory, product_name
FROM workspace.default.dim_products

  AND subcategory IS NOT NULL
```

## 📈 Sales Analysis

### Total Sales by Product Category
```sql
SELECT 
  p.category,
  SUM(f.sales_amount) AS total_sales
FROM workspace.default.fact_sales f
JOIN workspace.default.dim_products p
  ON f.product_key = p.product_key
GROUP BY p.category
ORDER BY total_sales DESC;
```

### Monthly Orders and Revenue Trend
```sql
SELECT 
  DATE_TRUNC('month', order_date) AS order_month,
  COUNT(DISTINCT order_number) AS total_orders,
  SUM(sales_amount) AS total_sales
FROM workspace.default.fact_sales
GROUP BY order_month
ORDER BY order_month;
```

## 👥 Customer Analytics

### Customer Age Group Order Distribution
```sql
SELECT 
    CASE 
        WHEN YEAR(CURRENT_DATE()) - YEAR(c.birthdate) BETWEEN 10 AND 20 THEN '10-20'
        WHEN YEAR(CURRENT_DATE()) - YEAR(c.birthdate) BETWEEN 21 AND 30 THEN '21-30'
        WHEN YEAR(CURRENT_DATE()) - YEAR(c.birthdate) BETWEEN 31 AND 40 THEN '31-40'
        ELSE '40+' 
    END AS age_group,
    COUNT(DISTINCT s.order_number) AS number_of_orders
FROM workspace.default.fact_sales s
JOIN workspace.default.dim_customers c 
    ON s.customer_key = c.customer_key
GROUP BY age_group;
```

### Top 10 Customers by Total Sales
```sql
SELECT 
    customer_key, 
    SUM(sales_amount) AS total_sales
FROM workspace.default.fact_sales
GROUP BY customer_key
ORDER BY total_sales DESC
LIMIT 10;
```

## 📉 Advanced Metrics

### Estimate Profit from Sales
```sql
SELECT *, (sales_amount * 0.30) AS profit
FROM workspace.default.fact_sales;
```

### Calculate Days Between Consecutive Orders
```sql
WITH CTE_TAB AS (
   SELECT 
    order_date AS currentdate,
    LAG(order_date) OVER (ORDER BY order_date) AS previousdate
    FROM workspace.default.fact_sales
)
SELECT *, currentdate - previousdate AS date_difference
FROM CTE_TAB;
```

### Running Total of Sales Over Time
```sql
SELECT 
  order_date,
  sales_amount,
  SUM(sales_amount) OVER (ORDER BY order_date) AS running_total
FROM workspace.default.fact_sales
ORDER BY order_date;
```

### Revenue Analysis by Day of Week
```sql
SELECT 
    DAYNAME(order_date) AS day_of_week,
    AVG(sales_amount) AS avg_daily_sales,
    SUM(sales_amount) AS total_revenue
FROM workspace.default.fact_sales
GROUP BY day_of_week, DAYOFWEEK(order_date)
ORDER BY DAYOFWEEK(order_date);
```

### Classify Orders by Sales Value
```sql
SELECT 
    order_number,
    sales_amount,
    CASE 
        WHEN sales_amount > 500 THEN 'Top Sales'
        WHEN sales_amount BETWEEN 100 AND 500 THEN 'Mid Sales'
        ELSE 'Low Sales'
    END AS sales_tier
FROM workspace.default.fact_sales
ORDER BY sales_amount DESC;
```

## 🧩 Data Quality & Insights

### Identify Products Without Sales
```sql
SELECT 
    p.product_name,
    p.category,
    p.subcategory
FROM workspace.default.dim_products p
LEFT JOIN workspace.default.fact_sales f 
    ON p.product_key = f.product_key
WHERE f.product_key IS NULL;
```

### Customer Purchase Activity Duration Analysis
```sql
SELECT 
    customer_key,
    MIN(order_date) AS first_buy,
    MAX(order_date) AS latest_buy,
    DATEDIFF(MAX(order_date), MIN(order_date)) AS days_active
FROM workspace.default.fact_sales
GROUP BY customer_key
HAVING days_active > 0
ORDER BY days_active DESC;
```
